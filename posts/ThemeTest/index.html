5. 代码块 (Code Blocks)

JavaScript:

/**
 * 这是一个 JavaScript 代码块
 */
function greet(name) {
    const message = `Hello, ${name}!`;
    console.log(message);
    return message;
}
greet('Typora');

Python:

# 这是一个 Python 代码块
def fibonacci(n):
    if n <= 1:
        return n
    else:
        return fibonacci(n-1) + fibonacci(n-2)

print([fibonacci(i) for i in range(10)])
